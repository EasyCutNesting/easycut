  19. ;function: STRPARSE
    20. 

  20. ;	This is a string parsing function - a helper routine
    21. 

  21. ;	you won't need to make any changes or customizations to this function
    22. 

  22. ;*************************************************************
    23. 

  23. ;*************************************************************
    24. 

  24. (defun strParse	(Str Delimiter / SearchStr StringLen return n char)
    25. 

  25. 	(setq SearchStr Str)
    26. 

  26. 	(setq StringLen (strlen SearchStr))
    27. 

  27. 	(setq return '())
    28. 

  28. 

  29. 	(while (> StringLen 0)
    30. 

  30. 		(setq n 1)
    31. 

  31. 		(setq char (substr SearchStr 1 1))
    32. 

  32. 		(while (and (/= char Delimiter) (/= char ""))
    33. 

  33. 			(setq n (1+ n))
    34. 

  34. 			(setq char (substr SearchStr n 1))
    35. 

  35. 		) ;_ end of while
    36. 

  36. 		(setq return (cons (substr SearchStr 1 (1- n)) return))
    37. 

  37. 		(setq SearchStr (substr SearchStr (1+ n) StringLen))
    38. 

  38. 		(setq StringLen (strlen SearchStr))
    39. 

  39. 	) ; end of while
    40. 

  40. 	(reverse return)
    41. 

  41. ); end function

  47. ;function: ADDTRUSTPATH
    48. 

  48. ;	This is the static function that defines how to handle the Path Locations you will specify later
    49. 

  49. ;	you won't need to make any changes or customizations to this function, either
    50. 

  50. ;*************************************************************
    51. 

  51. ;*************************************************************
    52. 

  52. (defun addTrustPath (dir pos / tmp c lst)
    53. 

  53.  (setq tmp "" c -1)
    54. 

  54.  (if (not (member (strcase dir)
    55. 

  55.  (setq lst (mapcar 'strcase (strParse (getvar "TRUSTEDPATHS") ";")))))
    56. 

  56.   (progn
    57. 

  57.    (if (not pos) (setq tmp (strcat dir ";" (getvar "TRUSTEDPATHS")))
    58. 

  58.      (mapcar '(lambda (x)
    59. 

  59.       (setq tmp (if (= (setq c (1+ c)) pos)
    60. 

  60.       (strcat tmp ";" dir ";" x)
    61. 

  61.       (strcat tmp ";" x)
    62. 

  62.      )
    63. 

  63.     )
    64. 

  64.    )
    65. 

  65.    lst
    66. 

  66.   )
    67. 

  67.  )
    68. 

  68.  (setvar "TRUSTEDPATHS" tmp)
    69. 

  69.  )
    70. 

  70.  )
    71. 

  71.  (princ)
    72. 

  72. ); end function

    85. 

  85. (addTrustPath "C:\\MY-MENU\\..." nil)

  106. ;function: REMOVETRUSTPATH
    107. 

  107. ;	This is the static function that defines how to handle the Path Locations you will specify later
    108. 

  108. ;	you won't need to make any changes or customizations to this function
    109. 

  109. ;*************************************************************
    110. 

  110. 

  111. (defun RemoveTrustPath (PathtoDelete / lst tmp)
    112. 

  112. (setq PathtoDelete (strcase PathtoDelete))
    113. 

  113. (setq lst (mapcar 'strcase (strParse (getvar "TRUSTEDPATHS") ";")))
    114. 

  114. (if (member PathtoDelete lst)
    115. 

  115. (progn
    116. 

  116. (setq lst (vl-remove (strcase PathToDelete) lst))
    117. 

  117. (setq tmp "")
    118. 

  118. (foreach folder lst
    119. 

  119. (setq tmp (strcat tmp folder ";"))
    120. 

  120. )
    121. 

  121. (setvar "TRUSTEDPATHS" tmp)
    122. 

  122. )
    123. 

  123. )
    124. 

  124. (princ)
    125. 

  125. ); end function
