(defun InsIns (EnameShape / SearchDownLevel RemoveDownLevelToOneLevel UpdateOneLevel
							LstEnameInside LstHimSelf itm EnamesHimSelf LevelOne NewLevelOne NewNewLevelOne LstDownLevel
							TmpLst LevelTwo Rtn)
	
	(defun SearchDownLevel (LevelOne LstHimSelf / itm Rtn)
		; LevelOne   (<Entity name: 28ba49f78c0> (<Entity name: 28ba49f7950> <Entity name: 28ba49f7930> <Entity name: 28ba49f78e0>))
		; LstHimSelf (<Entity name: 28ba49f7950> <Entity name: 28ba49f7930> <Entity name: 28ba49f78e0>)
		(if (and LevelOne LstHimSelf)
			(progn
				(setq Rtn T)
				(foreach itm (cadr LevelOne)
					(if (not (member itm LstHimSelf))
						(setq Rtn nil)
					)
				)
			)
		)
		Rtn
	)
	;
	(defun RemoveDownLevelToOneLevel (LevelOne LstDownLevel / itm Rtn)
	
		(if (and LevelOne LstDownLevel)
			(progn
				(foreach itm (cadr LevelOne)
					(if (not (member itm LstDownLevel))
						(setq Rtn (append Rtn (list itm)))
					)
				)
				(if Rtn 
					(setq Rtn (cons (car LevelOne) (list Rtn)))
					(setq Rtn LevelOne)
				)
			)
		)
		Rtn
	)
	;
	(defun UpdateOneLevel (LevelOne LstDownLevel / LevelOneCadr itm EnameChk Rtn)
		;LstDownLevel ((<Entity name: 28ba4a0ec10> (<Entity name: 28ba4a0ec50> <Entity name: 28ba4a0ec30>)) (<Entity name: 28ba4a0ebb0> (<Entity name: 28ba4a0ebf0> <Entity name: 28ba4a0ebd0>)))
		;LevelOne 	   (<Entity name: 28ba4a0eb40> (<Entity name: 28ba4a0ec10> <Entity name: 28ba4a0ebb0> <Entity name: 28ba4a0eb50>))
		
		(if (and LevelOne LstDownLevel)
			(progn
				(setq LevelOneCadr (cadr LevelOne)) 
				(foreach itm LstDownLevel
					(setq EnameChk (car itm)) 
					(if (member EnameChk LevelOneCadr)
						(setq LevelOneCadr (subst itm EnameChk LevelOneCadr)) 
					)
				)
				(setq Rtn (cons (car LevelOne) (list LevelOneCadr)))
			)
		)
		Rtn
	)
	;
	; Main
	;
	(if (and EnameShape)
		(progn
			(setq LstEnameInside (ShapeInside EnameShape))
			; search composed himself ++++++++++++++++++++++++++++++++
			(foreach itm LstEnameInside
				(if (null (setq EnamesHimSelf (ShapeInside itm)))
				 	(setq LstHimSelf (append LstHimSelf (list itm)))
				)
			)
			(princ "\n Se stessi ") (princ LstHimSelf) (princ "\n")
			; ++++++++++++++++++++++++++++++++++++++++++++++++++++++++
			; set first level [insieme primo livello]
			(foreach itm LstEnameInside
				(if (not (member itm LstHimSelf))
					(setq LevelOne (append LevelOne (list (list itm (ShapeInside itm)))))
				)
			)
			(setq LevelOne (append LevelOne (list (list EnameShape LstEnameInside))))
			(princ "\n 1° livello ") (princ LevelOne) (princ "\n")
			; ++++++++++++++++++++++++++++++++++++++++++++++++++++++++
			; filtro gli insiemi non scomponibili
			(setq NewLevelOne LevelOne)
			(foreach itm NewLevelOne
				(if (SearchDownLevel itm LstHimSelf)
					(progn
						(setq LstDownLevel   (append LstDownLevel (list itm)))
						(setq NewLevelOne    (LM:RemoveOnce itm NewLevelOne))
					)
				)
			)
			(princ "\n non scomponibili ") (princ LstDownLevel) (princ "\n")
			(princ "\n 1° livello nuovo ") (princ NewLevelOne)  (princ "\n")

			; ++++++++++++++++++++++++++++++++++++++++++++++++++++++++
			; elimino dal primo livello i non scomponibili
			(if NewLevelOne
				(progn
					(foreach itm LstDownLevel
						(setq TmpLst (append TmpLst (car (cdr itm))))
					)
					(princ "\n non scomponibili base ") (princ TmpLst) (princ "\n")
					(foreach itm NewLevelOne
						(setq NewNewLevelOne (append NewNewLevelOne (list (RemoveDownLevelToOneLevel itm TmpLst))))
					)
					(princ "\n 1° livello nuovo nuovo ") (princ NewNewLevelOne) (princ "\n")

					(foreach itm NewNewLevelOne
						(setq LevelTwo (append LevelTwo (list (UpdateOneLevel itm  LstDownLevel))))
					)
					(princ "\n 2° livello ") (princ LevelTwo) (princ "\n")
					
					; ++++++++++++++++++++++++++++++++++++++++++++++++++++++++
					; resolution system
					; ++++++++++++++++++++++++++++++++++++++++++++++++++++++++
					(setq Rtn (SystemResolution LevelTwo))
				)
				(setq Rtn LevelOne)
			)
		)
	)
	Rtn
)

	(defun RemoveEach (a l / Remove 
							 itm Rtn)
	
		(defun Remove (itm l)
			(cond 
				((null l) 
					nil
				)
				((atom (car l))
					(cond 
						((equal itm (car l))
							(Remove itm (cdr l)))
						(t 
							(cons (car l) (Remove itm (cdr l)))
						)
					)
				)
				(t 
					(cons (Remove itm (car l)) (Remove itm (cdr l)))
				)
			)
		)
		;
		; Main
		;
		(if (and a l)
			(progn
				(setq Rtn l)
				(foreach itm (LM:flatten a)
					(setq Rtn (Remove itm Rtn))
				)
			)
		)
		Rtn

	(defun RemoveExtraParent (Lst / Strip)

		; (((A)) ((D ((E F))))) -> ((A) (D (E F)))
		
		(defun Strip (lst)
			(if (or (null lst) (atom lst) (not (null (cdr lst))))
				lst
				(Strip (car lst))
			)
		)
		
		(cond 
			((or (null lst) (atom lst))
				lst
			)
			((null (Strip (car lst)))
				(RemoveExtraParent (cdr lst))
			)
			(t
				(Strip (cons (RemoveExtraParent (Strip (car lst))) (RemoveExtraParent (cdr lst))))
			)
		)
	)
