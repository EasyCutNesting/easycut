  97.   ;;---------------------------------------------------------------;;
  98.   ;; styles                                                        ;;
  99.   ;;---------------------------------------------------------------;;
  100.   ;; List of tags to enclose various code items, list must be in   ;;
  101.   ;; the following form:                                           ;;
  102.   ;;                                                               ;;
  103.   ;; (                                                             ;;
  104.   ;;   (<container open> <container close>)   e.g. (<pre> </pre>)  ;;
  105.   ;;   (<quote     open> <quote     close>)   quotes/dots          ;;
  106.   ;;   (<bracket   open> <bracket   close>)   parentheses          ;;
  107.   ;;   (<mcomment  open> <mcomment  close>)   multiline comments   ;;
  108.   ;;   (<scomment  open> <scomment  close>)   single comments      ;;
  109.   ;;   (<string    open> <string    close>)   strings              ;;
  110.   ;;   (<protected open> <protected close>)   protected symbols    ;;
  111.   ;;   (<integer   open> <integer   close>)   integers             ;;
  112.   ;;   (<real      open> <real      close>)   reals                ;;
  113.   ;; )                                                             ;;
  114.   ;;                                                               ;;
  115.   ;; If no tag is required for an item, the entry must still be    ;;
  116.   ;; present, as:                                                  ;;
  117.   ;;                                                               ;;
  118.   ;;   ("" "")                                                     ;;
  119.   ;;                                                               ;;
  120.   ;;---------------------------------------------------------------;;
  121.   ;; extn                                                          ;;
  122.   ;;---------------------------------------------------------------;;
  123.   ;; Extension for resultant file - dot must be included.          ;;
  124.   ;;                                                               ;;
  125.   ;; Example:   ".html"                                            ;;
  126.   ;;                                                               ;;
  127.   ;;---------------------------------------------------------------;;
  128.   ;; <>                                                            ;;
  129.   ;;---------------------------------------------------------------;;
  130.   ;; Boolean flag to determine whether the characters '<' and '>'  ;;
  131.   ;; are to be replaced with &lt; and &gt;                         ;;
  132.   ;;                                                               ;;

  134.   ;;                                                               ;;
  135.   ;;---------------------------------------------------------------;;
  136.  
  137.   ;;---------------------------------------------------------------;;

  139.   ;; type of code item, any user-defined symbols in the current    ;;
  140.   ;; namespace will be interpreted as the datatype to which they   ;;
  141.   ;; point.                                                        ;;

  146.   ;; functions or global variables.                                ;;

  234.   ;; Substitutes all instances of '<' and '>' with &lt; and &gt; in

  239.   (defun _fix<> ( string / _stringsubst )

  241.     (defun _stringsubst ( new old string / i nl )
  242.       (setq i 0 nl (strlen new))
  243.  
  244.       (while (and (< i (strlen string)) (setq i (vl-string-search old string i)))
  245.         (setq string (vl-string-subst new old string i) i (+ i nl))
  246.       )
  247.       string
  248.     )
  249.     
  250.     (_stringsubst "&lt;" "<" (_stringsubst "&gt;" ">" string))
  251.   )
